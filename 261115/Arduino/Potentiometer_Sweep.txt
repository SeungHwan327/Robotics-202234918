#include <ESP32Servo.h>

#define SERVO_PIN 14
const int potPin = 39;

Servo myServo;


void setup() {
    Serial.begin(115200);  
    myServo.attach(SERVO_PIN);
    myServo.write(0);
}

void loop() {
  int value = analogRead(potPin);

  Serial.println(value);
  

  for(int i=0; i<value; i++) {
  myServo.write(i);
  }
  for(int i=value; i>0; i--) {
  myServo.write(i);
  }
}
